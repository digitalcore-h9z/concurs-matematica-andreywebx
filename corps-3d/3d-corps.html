Arii si volume

Prisma triunghiulară

\( A_b = \dfrac{a \cdot h_a}{2} \)

\( A_l = P_b \cdot H \)

\( A_t = A_l + 2 \cdot A_b \)

\( V = A_b \cdot H \)

Prisma patrulateră

\( A_b = a \cdot b \)

\( A_l = P_b \cdot H \)

\( A_t = A_l + 2 \cdot A_b \)

\( V = A_b \cdot H \)

Cub

\( A_f = a^2 \)

\( A_t = 6a^2 \)

\( d_f = a\sqrt{2} \)

\( d = a\sqrt{3} \)

\( V = a^3 \)

Paralelipiped dreptunghic

\( A_t = 2(ab + bc + ca) \)

\( d = \sqrt{a^2 + b^2 + c^2} \)

\( V = a \cdot b \cdot c \)

Piramidă triunghiulară

\( A_b = \dfrac{a \cdot h_a}{2} \)

\( A_l = \dfrac{P_b \cdot a_p}{2} \)

\( A_t = A_l + A_b \)

\( V = \dfrac{A_b \cdot H}{3} \)

Piramidă patrulateră

\( A_b = a^2 \)

\( A_l = \dfrac{P_b \cdot a_p}{2} \)

\( A_t = A_l + A_b \)

\( V = \dfrac{A_b \cdot H}{3} \)

Cilindru

\( A_b = \pi r^2 \)

\( A_l = 2\pi r h \)

\( A_t = 2\pi r(r + h) \)

\( V = \pi r^2 h \)

Con

\( g = \sqrt{r^2 + h^2} \)

\( A_b = \pi r^2 \)

\( A_l = \pi r g \)

\( A_t = \pi r(r + g) \)

\( V = \dfrac{\pi r^2 h}{3} \)

Sferă

\( A = 4\pi r^2 \)

\( V = \dfrac{4\pi r^3}{3} \)
